# phone_call_effect.py tool. WAV/FLAC/OGG need nothing.
#   Windows: winget install --id Gyan.FFmpeg -e
#   macOS:   brew install ffmpeg
#   Linux:   sudo apt install ffmpeg

# export_onnx.py only. Not needed to train, and NOT what runs on the phone -
# the phone uses the onnxruntime-android Gradle dependency instead.
onnx>=1.15
onnxruntime>=1.17
